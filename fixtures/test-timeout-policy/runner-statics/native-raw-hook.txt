// Identity is the deadline-capture proof; a 400ms wall-clock sleep raced the following
// test's 250ms testTimeout on loaded CI (#2443). File-level collectors delete `.suite`.

  const collector = native.TestRunner.getCurrentSuite()
  collector.on('beforeAll', raw)
  native.expect(native.TestRunner.getSuiteHooks(collector.suite).beforeAll.at(-1)).toBe(raw)
  test('body runs after the uncapped raw hook', continued)
})
